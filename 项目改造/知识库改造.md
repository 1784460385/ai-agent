1. 任务背景与目标
当前问题：所有知识库文档混杂存放在同一目录，使用单一的 Chroma Collection 进行向量检索。当用户提问时，系统无法区分问题领域，可能从无关文档中检索到噪声，导致回答准确率下降。

改造目标：

数据层：知识库按“部门”分类存储，每个部门对应独立的目录和独立的 Chroma Collection。

检索层：实现“规则路由 + LLM 兜底”的双层路由机制。先通过关键词/正则快速匹配部门，未命中时再调用 DeepSeek 进行智能意图识别。

接口层：改造 /api/chat 接口，使其能够根据路由结果，在指定的一个或多个部门知识库中检索。

4.1 规则路由层（router.py 新增文件）

# router.py
import re
from typing import List, Optional
from config import DEPARTMENTS

def rule_based_route(query: str) -> Optional[List[str]]:
    """
    基于关键词和正则的快速路由。
    返回匹配到的部门列表；若无匹配，返回 None。
    """
    matched_departments = []

    for dept_key, dept_info in DEPARTMENTS.items():
        keywords = dept_info.get("keywords", [])
        for kw in keywords:
            # 使用正则表达式匹配，避免部分匹配问题（如 "工资" 不应匹配 "工资单" 之外无关内容）
            pattern = re.compile(re.escape(kw), re.IGNORECASE)
            if pattern.search(query):
                matched_departments.append(dept_key)
                break  # 该部门已命中，无需继续检查其他关键词

    if matched_departments:
        return matched_departments
    return None
AI 实现指令：

关键词匹配优先使用 re.escape 防止特殊字符导致正则异常。

支持多部门命中（如“对比人事和财务的请假流程”会同时匹配 hr 和 finance）。

匹配过程不区分大小写。

4.2 LLM 兜底路由层
python
# router.py (续)
import json
from openai import OpenAI

def llm_based_route(query: str, api_key: str, base_url: str, model: str) -> List[str]:
    """
    调用 DeepSeek 进行意图识别，返回部门列表。
    如果 LLM 也无法判断，返回所有部门。
    """
    # 构建部门描述文本
    dept_descriptions = "\n".join([
        f"- {key}: {info['name']}，{info['description']}"
        for key, info in DEPARTMENTS.items()
    ])

    system_prompt = f"""你是一个知识库路由助手。请根据用户的问题，判断该问题属于以下哪些部门。

可选的部门：
{dept_descriptions}

规则：
1. 如果问题明确属于某个部门，返回该部门的 key。
2. 如果问题涉及多个部门，返回多个 key（如 ["hr", "finance"]）。
3. 如果无法判断，返回所有部门的 key。
4. 你必须只返回一个 JSON 数组，格式为：{{"departments": ["key1", "key2"]}}，不要返回其他任何内容。

示例：
用户问题："工资怎么算？" → {{"departments": ["finance"]}}
用户问题："入职和请假流程分别是什么？" → {{"departments": ["hr"]}}
用户问题："今天天气怎么样？" → {{"departments": ["finance", "hr", "tech", "product"]}}
"""

    try:
        client = OpenAI(api_key=api_key, base_url=base_url)
        response = client.chat.completions.create(
            model=model,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": query}
            ],
            temperature=0.0,
            response_format={"type": "json_object"}  # 强制 JSON 输出
        )
        content = response.choices[0].message.content
        result = json.loads(content)
        departments = result.get("departments", [])
        # 校验返回的部门 key 是否合法
        valid_depts = [d for d in departments if d in DEPARTMENTS]
        return valid_depts if valid_depts else list(DEPARTMENTS.keys())
    except Exception as e:
        print(f"[WARN] LLM 路由失败: {e}，回退到全库检索")
        return list(DEPARTMENTS.keys())
AI 实现指令：

使用 response_format={"type": "json_object"} 强制 DeepSeek 输出结构化 JSON，避免解析失败。

如果 LLM 返回的 key 不合法，回退到全库检索。

如果 API 调用失败，同样回退到全库检索，确保系统可用性。

4.3 统一路由入口

# router.py (续)
def route_query(query: str, api_key: str, base_url: str, model: str) -> List[str]:
    """
    双层路由入口：先规则，后 LLM。
    """
    # 第一层：规则路由
    rule_result = rule_based_route(query)
    if rule_result:
        print(f"[ROUTER] 规则命中: {rule_result}")
        return rule_result

    # 第二层：LLM 兜底
    print(f"[ROUTER] 规则未命中，调用 LLM 兜底...")
    llm_result = llm_based_route(query, api_key, base_url, model)
    print(f"[ROUTER] LLM 路由结果: {llm_result}")
    return llm_result

# vector_store.py
import chromadb
from chromadb.utils import embedding_functions
from config import CHROMA_PERSIST_DIR, EMBEDDING_MODEL, DEPARTMENTS

def get_chroma_client():
    return chromadb.PersistentClient(path=CHROMA_PERSIST_DIR)

def get_embedding_function():
    return embedding_functions.SentenceTransformerEmbeddingFunction(
        model_name=EMBEDDING_MODEL
    )

def get_collection(dept: str):
    """获取指定部门的 Collection"""
    client = get_chroma_client()
    ef = get_embedding_function()
    return client.get_or_create_collection(
        name=f"kb_{dept}",
        embedding_function=ef
    )

def add_documents(dept: str, texts: list[str], ids: list[str]):
    """向指定部门写入文档"""
    collection = get_collection(dept)
    collection.add(documents=texts, ids=ids)
    return len(texts)

def query_documents(dept: str, query_text: str, n_results: int = 3):
    """从指定部门检索"""
    collection = get_collection(dept)
    results = collection.query(query_texts=[query_text], n_results=n_results)
    return results['documents'][0]

def query_multiple_departments(departments: list[str], query_text: str, n_results: int = 3):
    """从多个部门检索并合并结果"""
    all_docs = []
    for dept in departments:
        docs = query_documents(dept, query_text, n_results)
        all_docs.extend(docs)
    return all_docs

# rag_engine.py (修改 load_and_split_documents 和 index_knowledge_base)
import os
from langchain_community.document_loaders import PyPDFLoader, TextLoader, DirectoryLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from config import KNOWLEDGE_BASE_DIR, CHUNK_SIZE, CHUNK_OVERLAP, DEPARTMENTS

def load_and_split_documents():
    """遍历每个部门子目录，分别加载和分块"""
    dept_chunks = {}  # {dept: [chunks]}

    for dept in DEPARTMENTS.keys():
        dept_dir = os.path.join(KNOWLEDGE_BASE_DIR, dept)
        if not os.path.exists(dept_dir):
            print(f"[WARN] 部门目录不存在: {dept_dir}")
            continue

        all_docs = []
        for ext, loader_cls in [("**/*.pdf", PyPDFLoader), ("**/*.txt", TextLoader), ("**/*.md", TextLoader)]:
            try:
                loader = DirectoryLoader(dept_dir, glob=ext, loader_cls=loader_cls)
                all_docs.extend(loader.load())
            except Exception as e:
                print(f"[WARN] 加载 {dept_dir}/{ext} 失败: {e}")

        if not all_docs:
            continue

        splitter = RecursiveCharacterTextSplitter(
            chunk_size=CHUNK_SIZE,
            chunk_overlap=CHUNK_OVERLAP,
            separators=["\n\n", "\n", "。", "！", "？", " ", ""],
            length_function=len
        )
        chunks = splitter.split_documents(all_docs)
        dept_chunks[dept] = chunks

    return dept_chunks

def index_knowledge_base():
    """按部门分别索引"""
    from vector_store import add_documents
    dept_chunks = load_and_split_documents()
    if not dept_chunks:
        return {"status": "empty", "message": "知识库目录为空或无支持的文档"}

    total_indexed = 0
    for dept, chunks in dept_chunks.items():
        texts = [c.page_content for c in chunks]
        ids = [f"{dept}_chunk_{i}" for i in range(len(texts))]
        count = add_documents(dept, texts, ids)
        total_indexed += count
        print(f"[INDEX] {dept}: 索引了 {count} 个片段")

    return {"status": "ok", "chunks_indexed": total_indexed}

@app.post("/api/chat")
async def chat_endpoint(request: ChatRequest):
    """RAG 问答接口：路由 → 检索 → 增强 → 流式生成"""

    # Step 1: 双层路由，确定目标部门
    target_departments = route_query(
        query=request.message,
        api_key=request.config.api_key,
        base_url=request.config.base_url,
        model=request.config.model_name
    )

    # Step 2: 在目标部门中检索
    retrieved_docs = query_multiple_departments(
        departments=target_departments,
        query_text=request.message,
        n_results=RETRIEVAL_TOP_K
    )

    # Step 3: 构建增强 Prompt
    system_prompt = build_rag_prompt(request.message, retrieved_docs)

    # Step 4: 调用 DeepSeek
    client = OpenAI(
        api_key=request.config.api_key,
        base_url=request.config.base_url
    )
    stream = client.chat.completions.create(
        model=request.config.model_name,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": request.message}
        ],
        stream=True,
        temperature=0.3
    )

    # Step 5: SSE 流式返回
    async def event_stream():
        # 可选：先返回路由信息给前端展示
        yield f"data: [ROUTER] 命中部门: {', '.join(target_departments)}\n\n"
        for chunk in stream:
            delta = chunk.choices[0].delta
            if delta.content:
                yield f"data: {delta.content}\n\n"
        yield "data: [DONE]\n\n"

    return StreamingResponse(event_stream(), media_type="text/event-stream")

9. 注意事项
规则关键词维护：DEPARTMENTS 中的 keywords 需要根据实际业务持续迭代。建议初期由人工整理，后续可考虑从历史查询日志中自动挖掘高频词。

LLM 路由成本：LLM 兜底会增加一次额外的 API 调用。如果规则命中率较高（目标 >70%），整体成本可控。可在路由层加缓存，对相同问题复用路由结果。

多部门检索的排序：当从多个部门检索时，不同部门的文档可能相关度不同。后续可引入 Rerank 模型，对合并后的结果统一重排序。

Collection 命名规范：统一使用 kb_{dept} 格式，避免与 Chroma 中其他 Collection 冲突。

兼容性：改造后，原有的单库数据需要迁移。可以提供一个迁移脚本，将旧 Collection 中的文档按内容重新分配到对应部门，或直接清空重建索引。
